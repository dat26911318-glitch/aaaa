            $("#dangnhap").click(function() {
                let user = $("#username").val();
                let pass = $("#password").val();
                if (user == "" || pass == "") {
                    $("#thongbao").text("Vui lòng nhập đầy đủ thông tin!")
                                 .css("color", "red");
                }
                else if (user == "admin" && pass == "123456") {
                    $("#thongbao").text("Đăng nhập thành công!")
                                 .css("color", "green");
                }
                else {
                    $("#thongbao").text("Sai tên đăng nhập hoặc mật khẩu!")
                                 .css("color", "red");
                }

            });
